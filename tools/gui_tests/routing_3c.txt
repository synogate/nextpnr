# Phase 3c: route dock (candidates, session pips, arcs), waypoint routing (dock "Route to", Ctrl+click,
# arc double-click), Route arc / Route rest of net, Lock / Unlock routing; the result is checked in Python
# (tree connected, every sink wire of the router's routing reached) and saved as JSON.
# design: counter_pnr
resize 1600 1000
# (the net from the adder's OUT2 on X78Y116 through a switch box and the input mux IM.P08 to IN4 of the LUT on
# X77Y116, sink pin wire X77Y116/CPE.IN8_int)
py N="$auto$alumacc.cc:512:replace_alu$26.Y[1]"; nw=lambda: len(ctx.nets[N].wires); T0={str(w):m.pip for w,m in ctx.nets[N].wires}; print(len(T0), T0)
py S=[w for w in T0 if not T0[w]][0]; L0=["X77Y116/CPE.IN8_int"]; print("source", S, "sinks", L0, [w for w in T0 if not any(p and ctx.getPipSrcWire(p)==w for p in T0.values())])
py tree=lambda: {str(w):m.pip for w,m in ctx.nets[N].wires}
py ok=lambda: (lambda T: all(w==S or (T[w] and ctx.getPipSrcWire(T[w]) in T and ctx.getPipDstWire(T[w])==w and ctx.getBoundWireNet(w).name==N) for w in T) and all(l in T for l in L0))(tree())
py for u in ctx.nets[N].users: print("user", u.cell.name, u.port, u.cell.bel)
assert_py ok()
key E
# --- rip the net up, start route mode: the dock shows the head and its candidates
zoombel X77Y116/CPE_LT_L 1.2
clickpip X77Y116/IM.P08.Y/IM.P08.D1/INV right
menu Unroute net
wait 300
assert_py nw() <= 1
selectnet $auto$alumacc.cc:512:replace_alu$26.Y[1]
key R
wait 400
dock cands
dock sinks
zoomwire X78Y116/CPE.COMBOUT2_int 0.6
wait 300
shot 01_dock.png
# --- bind the first pip from the dock (the source wire has no graphics in the view)
dock cands X78Y116/CPE.OUT2_int
wait 200
assert_py nw() == 2
assert_status from X78Y116/CPE.OUT2_int
dock steps
# --- waypoint by name: the shortest free path from the head to the input mux of the sink CPE
settext routeTarget X77Y116/IM.P08.Y
button Route to
wait 300
status
# (X77Y116/IM.P08.Y is an alias of the node X76Y115/IM.P08.D5_n)
assert_status Route $auto$alumacc.cc:512:replace_alu$26.Y[1] to X76Y115/IM.P08.D5_n: 6 pips
assert_status from X76Y115/IM.P08.D5_n
py print(nw(), sorted(tree()))
dock steps
zoombel X77Y116/CPE_LT_L 1.2
wait 300
shot 02_waypoint.png

# --- a bridge net (its source wire is driven by a pip of another net) with 3 sinks: two LUT inputs and a CP-line cell: Unroute net,
# then Route rest of net from the wire menu routes all arcs as one undo entry
py import re; B="$abc$348$iopadmap$rst$bridge0$muxout"; TB={str(w):m.pip for w,m in ctx.nets[B].wires}; LB=[w for w in TB if re.search(r"CPE\.IN\d+_int$", w)]; print(len(TB), "sinks", len(LB), LB)
# (one root: the source wire, driven by a pip of the CP-line net; every other wire driven from a wire of the net)
py okb=lambda: (lambda T: all(ctx.getBoundWireNet(w).name==B for w in T) and sum(1 for w in T if not T[w] or ctx.getPipSrcWire(T[w]) not in T) == 1 and all(l in T for l in LB))({str(w):m.pip for w,m in ctx.nets[B].wires})
assert_py okb()
zoombel X77Y128/CPE_LT_U 0.6
clickpip X77Y128/CPE.IN2_int/CPE.IN2 right
menu Unroute net
wait 300
assert_py not okb()
py print(len(ctx.nets[B].wires))
zoombel X79Y128/CPE_LT_L 0.6
selectnet $abc$348$iopadmap$rst$bridge0$muxout
key R
wait 300

wait 500
status
assert_status 3 arcs routed (3/3 arcs routed, 0 conflicts)
assert_py okb()
py print(len(ctx.nets[B].wires), "wires (router:", len(TB), ")")
key Esc
zoombel X78Y127/CPE_LT_L 3
