# Phase 2b: drag & drop (move, swap) + undo, "Unplaced only" filter, Next unplaced (N), Move selection by dialog
# (legal / illegal offsets), arrow-key nudging. (In this design each LUT forms a cluster with the register it feeds:
# the register moves along, a cluster only moves to free bels; the drag swap uses new unclustered cells.)
# design: counter_pnr
resize 1400 1000
zoombel X77Y116/CPE_LT_L 1.2
py L1="$abc$348$auto$blifparse.cc:397:parse_blif$383"; L2="$abc$348$auto$blifparse.cc:397:parse_blif$381"; F1="$auto$ff.cc:337:slice$107"; at=lambda n: str(ctx.cells[n].bel)
key E
# --- drag to a free bel, undo, redo
dragbel X77Y116/CPE_LT_L X76Y116/CPE_LT_L
wait 200
assert_py at(L1) == "X76Y116/CPE_LT_L" and at(F1) == "X76Y116/CPE_FF_L"
key Ctrl+Z
assert_py at(L1) == "X77Y116/CPE_LT_L" and at(F1) == "X77Y116/CPE_FF_L"
key Ctrl+Y
assert_py at(L1) == "X76Y116/CPE_LT_L"
key Ctrl+Z
# --- drag onto an occupied bel: refused for a cluster, nothing moves
dragbel X77Y116/CPE_LT_L X77Y117/CPE_LT_L
wait 200
assert_status a cluster can only be moved to free bels
assert_py at(L1) == "X77Y116/CPE_LT_L" and at(L2) == "X77Y117/CPE_LT_L"
shot 01_drag_refused.png
# --- drag a single cell onto another swaps (two new unclustered cells like L1, placed by click)
selectcell $abc$348$auto$blifparse.cc:397:parse_blif$383
treemenu
menu Logic > New cell like
dialog sw_a
wait 300
clickbel X76Y118/CPE_LT_L
wait 200
selectcell $abc$348$auto$blifparse.cc:397:parse_blif$383
treemenu
menu Logic > New cell like
dialog sw_b
wait 300
clickbel X76Y119/CPE_LT_L
wait 200
key Esc
dragbel X76Y118/CPE_LT_L X76Y119/CPE_LT_L
wait 200
assert_status Swap
assert_py at("sw_a") == "X76Y119/CPE_LT_L" and at("sw_b") == "X76Y118/CPE_LT_L"
shot 01b_drag_swap.png
key Ctrl+Z
assert_py at("sw_a") == "X76Y118/CPE_LT_L" and at("sw_b") == "X76Y119/CPE_LT_L"
# (take the two cells out again: undo their placements and creation)
key Ctrl+Z
key Ctrl+Z
key Ctrl+Z
key Ctrl+Z
wait 300
assert_py "sw_a" not in ctx.cells and "sw_b" not in ctx.cells
# --- unplace via the bel menu, "Unplaced only", N arms it, click places it
key Esc
clickbel X77Y116/CPE_LT_L right
menu Unplace
wait 200
assert_py ctx.cells[L1].bel is None
check Unplaced only on
wait 300
shot 02_unplaced_only.png
key N
assert_status Placing cluster $abc$348$auto$blifparse.cc:397:parse_blif$383
zoombel X77Y116/CPE_LT_L 1.2
clickbel X76Y116/CPE_LT_L
wait 200
assert_status Place cluster $abc$348$auto$blifparse.cc:397:parse_blif$383 at X76Y116/CPE_LT_L
assert_py at(L1) == "X76Y116/CPE_LT_L"
key N
assert_status All cells are placed
shot 03_placed_again.png
check Unplaced only off
key Ctrl+Z
key Ctrl+Z
assert_py at(L1) == "X77Y116/CPE_LT_L"
# --- Move selection by: two cells, legal offset (-1, 0), then an illegal one (onto cells not moved)
key Esc
selectcell $abc$348$auto$blifparse.cc:397:parse_blif$383
selectcell $abc$348$auto$blifparse.cc:397:parse_blif$381 ctrl
assert_nostatus Moving
key M
dialog -1 0
wait 300
status
assert_py at(L1) == "X76Y116/CPE_LT_L" and at(L2) == "X76Y117/CPE_LT_L"
shot 04_moved_by.png
key M
dialog 3 0
wait 300
status
assert_status not moved
assert_py at(L1) == "X76Y116/CPE_LT_L" and at(L2) == "X76Y117/CPE_LT_L"
key M
dialog_cancel
# --- arrow keys nudge the selection by the smallest legal step (Up: up on the screen = larger y)
focus view
key Up
wait 300
status
py print(at(L1), at(L2))
assert_py at(L1) == "X76Y117/CPE_LT_L" and at(L2) == "X76Y118/CPE_LT_L"
key Ctrl+Z
assert_py at(L1) == "X76Y116/CPE_LT_L"
key Right
wait 300
status
py print(at(L1), at(L2))
assert_py at(L1) == "X77Y116/CPE_LT_L" and at(L2) == "X77Y117/CPE_LT_L"
shot 05_nudged.png
